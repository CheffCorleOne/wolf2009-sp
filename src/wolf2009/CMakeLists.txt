cmake_minimum_required(VERSION 3.12)
project(wolf2009_archive LANGUAGES C CXX)

# This project is intentionally independent of the engine/editor/game modules.
set(WOLF2009_MINIZ_SOURCE_DIR "" CACHE PATH "Optional upstream miniz directory containing miniz.c/miniz.h")
if(WOLF2009_MINIZ_SOURCE_DIR)
    add_library(wolf2009_miniz STATIC "${WOLF2009_MINIZ_SOURCE_DIR}/miniz.c")
    target_include_directories(wolf2009_miniz PUBLIC "${WOLF2009_MINIZ_SOURCE_DIR}")
    target_compile_definitions(wolf2009_miniz PUBLIC WOLF2009_USE_MINIZ)
    set(wolf2009_compression wolf2009_miniz)
else()
    find_package(ZLIB REQUIRED)
    set(wolf2009_compression ZLIB::ZLIB)
endif()

add_library(wolf2009_archive STATIC Archive.cpp)
target_compile_features(wolf2009_archive PUBLIC cxx_std_11)
target_include_directories(wolf2009_archive PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(wolf2009_archive PUBLIC ${wolf2009_compression})
add_executable(wolf2009-inspect Inspect.cpp SelfTest.cpp)
target_link_libraries(wolf2009-inspect PRIVATE wolf2009_archive)
enable_testing()
add_test(NAME wolf2009_archive_selftest COMMAND wolf2009-inspect --self-test)

add_library(wolf2009_geometry STATIC Geometry.cpp)
target_compile_features(wolf2009_geometry PUBLIC cxx_std_11)
target_include_directories(wolf2009_geometry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(wolf2009-scene SceneExport.cpp GeometryTest.cpp)
target_link_libraries(wolf2009-scene PRIVATE wolf2009_geometry wolf2009_archive)
add_test(NAME wolf2009_geometry_selftest COMMAND wolf2009-scene --self-test)
