cmake_minimum_required(VERSION 3.12)
project(wolf2009_archive LANGUAGES C CXX)

# This project is intentionally independent of the engine/editor/game modules.
set(WOLF2009_MINIZ_SOURCE_DIR "" CACHE PATH "Optional upstream miniz directory containing miniz.c/miniz.h")
if(WOLF2009_MINIZ_SOURCE_DIR)
    add_library(wolf2009_miniz STATIC "${WOLF2009_MINIZ_SOURCE_DIR}/miniz.c")
    target_include_directories(wolf2009_miniz PUBLIC "${WOLF2009_MINIZ_SOURCE_DIR}")
    target_compile_definitions(wolf2009_miniz PUBLIC WOLF2009_USE_MINIZ)
    set(wolf2009_compression wolf2009_miniz)
else()
    find_package(ZLIB REQUIRED)
    set(wolf2009_compression ZLIB::ZLIB)
endif()

add_library(wolf2009_archive STATIC Archive.cpp StreamMounts.cpp)
target_compile_features(wolf2009_archive PUBLIC cxx_std_11)
target_include_directories(wolf2009_archive PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(wolf2009_archive PUBLIC ${wolf2009_compression})
add_executable(wolf2009-inspect Inspect.cpp SelfTest.cpp)
target_link_libraries(wolf2009-inspect PRIVATE wolf2009_archive)
enable_testing()
add_test(NAME wolf2009_archive_selftest COMMAND wolf2009-inspect --self-test)

add_library(wolf2009_geometry STATIC Geometry.cpp)
target_compile_features(wolf2009_geometry PUBLIC cxx_std_11)
target_include_directories(wolf2009_geometry PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(wolf2009-scene SceneExport.cpp GeometryTest.cpp)
target_link_libraries(wolf2009-scene PRIVATE wolf2009_geometry wolf2009_archive)
add_test(NAME wolf2009_geometry_selftest COMMAND wolf2009-scene --self-test)

add_library(wolf2009_content STATIC Dds.cpp Entities.cpp Material.cpp SkinCatalog.cpp)
target_compile_features(wolf2009_content PUBLIC cxx_std_11)
target_include_directories(wolf2009_content PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(wolf2009_player_start STATIC PlayerStart.cpp)
target_link_libraries(wolf2009_player_start PUBLIC wolf2009_geometry wolf2009_content)
add_library(wolf2009_brain STATIC BrainGraph.cpp)
target_compile_features(wolf2009_brain PUBLIC cxx_std_11)
target_include_directories(wolf2009_brain PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(wolf2009-brains BrainInspect.cpp BrainGraphTest.cpp)
target_link_libraries(wolf2009-brains PRIVATE wolf2009_brain wolf2009_archive)
add_test(NAME wolf2009_brain_selftest COMMAND wolf2009-brains --self-test)
add_library(wolf2009_physics STATIC MapPhysics.cpp StaticCollision.cpp ConvexContact.cpp PlayerMove.cpp LinearMover.cpp MovingSupport.cpp BrainStartup.cpp)
target_link_libraries(wolf2009_physics PUBLIC wolf2009_player_start wolf2009_archive wolf2009_brain)
add_library(wolf2009_placement STATIC ModelPlacement.cpp)
target_link_libraries(wolf2009_placement PUBLIC wolf2009_content wolf2009_geometry wolf2009_archive)
target_sources(wolf2009_physics PRIVATE WagonScene.cpp TrainArrival.cpp DoorAnimation.cpp DoorModel.cpp TrainDoorSignal.cpp GreeterAnimation.cpp GreeterModel.cpp TrainStopScene.cpp RiderMovement.cpp)
target_link_libraries(wolf2009_physics PUBLIC wolf2009_placement)
target_sources(wolf2009-scene PRIVATE PlayerStartTest.cpp ModelPlacementTest.cpp)
target_link_libraries(wolf2009-scene PRIVATE wolf2009_physics)
target_link_libraries(wolf2009-scene PRIVATE wolf2009_placement)
add_executable(wolf2009-walk Walk.cpp StaticCollisionTest.cpp PlayerMoveTest.cpp LinearMoverTest.cpp MovingSupportTest.cpp ModelCollisionTest.cpp ConvexContactTest.cpp ModelConvexCollisionTest.cpp ModelPrimitiveCollisionTest.cpp BrainStartupTest.cpp WagonSceneTest.cpp TrainArrivalTest.cpp TrainDoorSignalTest.cpp DoorModelTest.cpp GreeterAnimationTest.cpp GreeterModelTest.cpp RiderMovementTest.cpp)
target_link_libraries(wolf2009-walk PRIVATE wolf2009_physics)
add_test(NAME wolf2009_walk_selftest COMMAND wolf2009-walk --self-test)
set(WOLF2009_TRAINYARD_MPK "" CACHE FILEPATH "Owner-supplied trainyard.mpk for the parked wagon contact test")
if(WOLF2009_TRAINYARD_MPK)
    add_executable(wolf2009-wagon-contacts ../../tests/WagonContacts.cpp)
    target_link_libraries(wolf2009-wagon-contacts PRIVATE wolf2009_physics)
    add_test(NAME wolf2009_wagon_contacts COMMAND wolf2009-wagon-contacts "${WOLF2009_TRAINYARD_MPK}")
    add_executable(wolf2009-door-animation ../../tests/TrainDoorAnimation.cpp)
    target_link_libraries(wolf2009-door-animation PRIVATE wolf2009_physics)
    add_test(NAME wolf2009_door_animation COMMAND wolf2009-door-animation "${WOLF2009_TRAINYARD_MPK}")
endif()
set(WOLF2009_SP_BASE "" CACHE PATH "Owner-supplied SP/base folder for the trainyard gameplay test")
if(WOLF2009_SP_BASE)
    add_executable(wolf2009-train-stop ../../tests/TrainStopScene.cpp)
    target_link_libraries(wolf2009-train-stop PRIVATE wolf2009_physics)
    add_test(NAME wolf2009_train_stop COMMAND wolf2009-train-stop "${WOLF2009_SP_BASE}")
endif()
add_library(wolf2009_zip STATIC ZipArchive.cpp)
target_compile_features(wolf2009_zip PUBLIC cxx_std_11)
target_include_directories(wolf2009_zip PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(wolf2009_zip PUBLIC ${wolf2009_compression})
target_sources(wolf2009_content PRIVATE DeclarationText.cpp ScriptedVoice.cpp)
target_link_libraries(wolf2009_content PUBLIC wolf2009_zip wolf2009_archive)
add_executable(wolf2009-content ContentTest.cpp DdsTest.cpp EntitiesTest.cpp MaterialTest.cpp SkinCatalogTest.cpp ZipTest.cpp StreamMountsTest.cpp ScriptedVoiceTest.cpp)
target_link_libraries(wolf2009-content PRIVATE wolf2009_content wolf2009_zip wolf2009_archive)
add_test(NAME wolf2009_content_selftest COMMAND wolf2009-content --self-test)
add_executable(wolf2009-pack ../../tools/scene-pack.cpp)
target_link_libraries(wolf2009-pack PRIVATE wolf2009_content wolf2009_geometry wolf2009_archive wolf2009_zip)
