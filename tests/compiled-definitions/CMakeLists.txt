cmake_minimum_required(VERSION 3.12)
project(wolf2009_compiled_definitions_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(core "${CMAKE_CURRENT_SOURCE_DIR}/../../src/wolf2009")
if(NOT TARGET wolf2009_compiled_declarations)
    add_library(wolf2009_compiled_declarations STATIC "${core}/CompiledDeclarations.cpp")
    target_include_directories(wolf2009_compiled_declarations PUBLIC "${core}")
    target_compile_features(wolf2009_compiled_declarations PUBLIC cxx_std_11)
endif()
if(NOT TARGET wolf2009_compiled_catalog)
    add_library(wolf2009_compiled_catalog STATIC "${core}/CompiledDeclarationCatalog.cpp")
    target_include_directories(wolf2009_compiled_catalog PUBLIC "${core}")
    target_compile_features(wolf2009_compiled_catalog PUBLIC cxx_std_11)
    target_link_libraries(wolf2009_compiled_catalog PUBLIC wolf2009_compiled_declarations)
endif()
if(NOT TARGET wolf2009_source_definitions)
    add_library(wolf2009_source_definitions STATIC "${core}/SourceDefinitions.cpp")
    target_include_directories(wolf2009_source_definitions PUBLIC "${core}")
    target_compile_features(wolf2009_source_definitions PUBLIC cxx_std_11)
endif()
set(WOLF2009_MINIZ_SOURCE_DIR "" CACHE PATH "Optional existing pinned miniz source directory")
if(WOLF2009_MINIZ_SOURCE_DIR)
    add_library(compiled_definitions_miniz STATIC "${WOLF2009_MINIZ_SOURCE_DIR}/miniz.c")
    target_include_directories(compiled_definitions_miniz PUBLIC "${WOLF2009_MINIZ_SOURCE_DIR}")
    set(compression compiled_definitions_miniz)
else()
    find_package(ZLIB REQUIRED)
    set(compression ZLIB::ZLIB)
endif()
add_library(compiled_definitions_archive STATIC "${core}/Archive.cpp")
target_include_directories(compiled_definitions_archive PUBLIC "${core}")
target_link_libraries(compiled_definitions_archive PUBLIC ${compression})
if(WOLF2009_MINIZ_SOURCE_DIR)
    target_compile_definitions(compiled_definitions_archive PRIVATE WOLF2009_USE_MINIZ)
endif()
add_executable(wolf2009-compiled-definitions CompiledDefinitionsTest.cpp)
target_link_libraries(wolf2009-compiled-definitions PRIVATE wolf2009_source_definitions wolf2009_compiled_catalog compiled_definitions_archive)
add_test(NAME wolf2009_compiled_definitions_synthetic COMMAND wolf2009-compiled-definitions --self-test)
set(WOLF2009_COMPILED_DEFINITIONS_SOURCE_MAP "" CACHE FILEPATH "Optional owner-supplied selected trainyard MPK")
if(WOLF2009_COMPILED_DEFINITIONS_SOURCE_MAP)
    add_test(NAME wolf2009_compiled_definitions_original COMMAND wolf2009-compiled-definitions --source-map "${WOLF2009_COMPILED_DEFINITIONS_SOURCE_MAP}")
endif()
