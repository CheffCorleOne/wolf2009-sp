cmake_minimum_required(VERSION 3.12)
project(wolf2009_source_rig_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(core "${CMAKE_CURRENT_SOURCE_DIR}/../../src/wolf2009")
add_library(source_rig_animation STATIC "${core}/SourceRigAnimation.cpp"
    "${core}/SplineClipData.cpp" "${core}/SplineTransformCodec.cpp" "${core}/SkeletonData.cpp")
target_include_directories(source_rig_animation PUBLIC "${core}")
if(NOT MSVC)
    target_compile_options(source_rig_animation PRIVATE -ffp-contract=off -fno-fast-math)
endif()
add_executable(wolf2009-source-rig SourceRigAnimationTest.cpp)
target_link_libraries(wolf2009-source-rig PRIVATE source_rig_animation)
enable_testing()
add_test(NAME wolf2009_source_rig_synthetic COMMAND wolf2009-source-rig)
set(WOLF2009_SP_BASE "" CACHE PATH "Optional owner-supplied read-only SP/base")
set(WOLF2009_MINIZ_SOURCE_DIR "" CACHE PATH "Optional pinned miniz source directory")
if(WOLF2009_SP_BASE)
    if(WOLF2009_MINIZ_SOURCE_DIR)
        add_library(source_rig_miniz STATIC "${WOLF2009_MINIZ_SOURCE_DIR}/miniz.c")
        target_include_directories(source_rig_miniz PUBLIC "${WOLF2009_MINIZ_SOURCE_DIR}")
        set(compression source_rig_miniz)
    else()
        find_package(ZLIB REQUIRED)
        set(compression ZLIB::ZLIB)
    endif()
    add_executable(wolf2009-source-rig-original SourceCompatibility.cpp
        "${core}/Archive.cpp" "${core}/StreamMounts.cpp" "${core}/GreeterAnimation.cpp")
    target_link_libraries(wolf2009-source-rig-original PRIVATE source_rig_animation ${compression})
    if(WOLF2009_MINIZ_SOURCE_DIR)
        target_compile_definitions(wolf2009-source-rig-original PRIVATE WOLF2009_USE_MINIZ)
    endif()
    add_test(NAME wolf2009_source_rig_original COMMAND wolf2009-source-rig-original --source-base "${WOLF2009_SP_BASE}")
endif()
