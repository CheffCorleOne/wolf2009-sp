cmake_minimum_required(VERSION 3.12)
project(wolf2009_source_weapon_checks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(core "${CMAKE_CURRENT_SOURCE_DIR}/../../src/wolf2009")
add_library(source_weapon_profile STATIC "${core}/SourceWeaponProfile.cpp"
    "${core}/SourceDefinitions.cpp" "${core}/CompiledDeclarations.cpp"
    "${core}/CompiledDeclarationCatalog.cpp" "${core}/CompiledDeclarationNamespace.cpp")
target_include_directories(source_weapon_profile PUBLIC "${core}")
set(WOLF2009_MINIZ_SOURCE_DIR "" CACHE PATH "Optional existing pinned miniz directory")
if(WOLF2009_MINIZ_SOURCE_DIR)
    add_library(weapon_compression STATIC "${WOLF2009_MINIZ_SOURCE_DIR}/miniz.c")
    target_include_directories(weapon_compression PUBLIC "${WOLF2009_MINIZ_SOURCE_DIR}")
    target_compile_definitions(weapon_compression PUBLIC WOLF2009_USE_MINIZ)
else()
    find_package(ZLIB REQUIRED)
    add_library(weapon_compression INTERFACE)
    target_link_libraries(weapon_compression INTERFACE ZLIB::ZLIB)
endif()
add_library(weapon_archives STATIC "${core}/Archive.cpp" "${core}/ZipArchive.cpp")
target_include_directories(weapon_archives PUBLIC "${core}")
target_link_libraries(weapon_archives PUBLIC weapon_compression)
add_executable(wolf2009-source-weapon SourceWeaponTest.cpp)
target_link_libraries(wolf2009-source-weapon PRIVATE source_weapon_profile weapon_archives)
enable_testing()
add_test(NAME wolf2009_source_weapon_synthetic COMMAND wolf2009-source-weapon --self-test)
set(WOLF2009_SOURCE_WEAPON_SP_BASE "" CACHE PATH "Optional owner-supplied original SP/base")
if(WOLF2009_SOURCE_WEAPON_SP_BASE)
    add_test(NAME wolf2009_source_weapon_original COMMAND wolf2009-source-weapon --source-base "${WOLF2009_SOURCE_WEAPON_SP_BASE}")
endif()
