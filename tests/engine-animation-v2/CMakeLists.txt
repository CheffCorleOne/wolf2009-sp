cmake_minimum_required(VERSION 3.12)
project(wolf2009_animation_wire_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(adapter "${CMAKE_CURRENT_SOURCE_DIR}/../../src/idtech4a_anim")
add_executable(wolf2009-animation-wire-v2 AnimationWireV2Test.cpp "${adapter}/AnimationWireV2.cpp")
target_include_directories(wolf2009-animation-wire-v2 PRIVATE "${adapter}")
add_executable(wolf2009-animation-wire-v1-control
    "${CMAKE_CURRENT_SOURCE_DIR}/../engine-animation/AnimationWireTest.cpp" "${adapter}/AnimationWire.cpp")
target_include_directories(wolf2009-animation-wire-v1-control PRIVATE "${adapter}")
foreach(target wolf2009-animation-wire-v2 wolf2009-animation-wire-v1-control)
    if(NOT MSVC)
        target_compile_options(${target} PRIVATE -ffp-contract=off -fno-fast-math)
    endif()
endforeach()
enable_testing()
add_test(NAME wolf2009_animation_wire_v2 COMMAND wolf2009-animation-wire-v2)
add_test(NAME wolf2009_animation_wire_v1_control COMMAND wolf2009-animation-wire-v1-control)
